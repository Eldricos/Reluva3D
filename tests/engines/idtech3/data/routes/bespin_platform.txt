# bespin_platform's way out, for sp-run -tour 5000 -replay (set helpUsObi
# 1). The tour leaves out the console that unlocks the platform doors, as
# it also shows them through a misc_camera; here, from the tour's end:
# the console (use, facing north), and use again to leave the camera's
# view (in it, the player does nothing). The doors' unlocking switches on
# the trigger for cinematic 22, at whose end Tavion comes; she blocks a
# saber and blaster bolts, and mostly rockets and the disruptor too, so
# the flechette does most of the work: how long she lasts turns on the
# game's random draws. Her death script is cinematic 23, which changes
# the level.
/setviewpos 1640 1488 440 90
@until +300 0 0
@until +500 32 0
@until +6000 0 0
@until +300 32 0
@until +2000 0 0
# cinematic 22
/setviewpos 2440 2004 24 90
@until +85000 0 0
# Tavion
@repeat 100
@nearest real_tavion +160
@until +1000 1 9
@nearest real_tavion
@until +1000 1 8
@nearest real_tavion
@until +1000 1 4
@nearest real_tavion
@until +1000 1 8
@end
# cinematic 23
@until +90000 0 0
